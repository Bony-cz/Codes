#include <iostream>
#include <unistd.h>
using namespace std;

void logo() {
    cout<<"========================================" <<endl;
    cout<<"=                                      =" <<endl;
    cout<<"=   BBBB   N   N  Y   Y   OOO   SSSS   =" <<endl;
    cout<<"=   B   B  NN  N   Y Y   O   O  S      =" <<endl;
    cout<<"=   BBBB   N N N    Y    O   O   SSS   =" <<endl;
    cout<<"=   B   B  N  NN    Y    O   O      S  =" <<endl;
    cout<<"=   BBBB   N   N    Y     OOO   SSSS   =" <<endl;
    cout<<"=                                      =" <<endl;
    cout<<"========================================" <<endl;
}

void start() {
    cout<<"Press Enter...";
    cin.get();
    sleep(1);
    cout<<"Welcome to BonyOS" <<endl;
    logo();
    cout<<""<<endl;
    cout<<""<<endl;
}

int main()
{
    start();
    string name;
    cout<<"Insert name: ";
    cin>>name;
    cout<<"" <<endl;

    bool status = true;
    string command;

    while (status) {

        cout<<name <<"/pc/-$";
        cin>>command;

        if (command == "help") {
            cout<<"help" <<endl;
            cout<<"creator" <<endl;
            cout<<"echo" <<endl;
            cout<<"neofetch" <<endl;
            cout<<"reboot" <<endl;
        }

        else if (command == "creator") {
            cout<<"Bony__88" <<endl;
        }

        else if (command == "neofetch") {
            logo();
            cout<<"          CPU:Intel I5-14400f" <<endl;
            cout<<"          GPU:Nvidia Rtx-5060" <<endl;
            cout<<"                  SSD:1t" <<endl;
            cout<<"                 Ram:32GB" <<endl;
        }

        else if (command == "reboot") {
            sleep(0.5);
            start();
        }

        else if (command == "echo") {
            string echo;

            cout<<">";
            cin>>echo;

            cout<<echo <<endl;
        }

        else {
            cout<<command <<" not recognised" <<endl;
        }
    }

    return 0;
}
